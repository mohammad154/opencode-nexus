+def validate_single_symbol_rule(
+    rule: dict,
+    expected_symbol: str | None = None,
+) -> str:
+    """Validate single-symbol provenance; return the symbol or fail closed.
+
+    Raises ``ValueError`` when provenance is missing, when the rule spans
+    zero or multiple symbols, or when *expected_symbol* is given and the
+    rule belongs to a different island (cross-symbol migration in v1).
+    """
+    if not isinstance(rule, dict):
+        raise ValueError(
+            "validate_single_symbol_rule: rule must be a dict; "
+            "fail closed, refusing an unprovenanced rule."
+        )
+    provenance = rule.get("provenance")
+    if not isinstance(provenance, dict) or not str(
+        provenance.get("symbol", "")
+    ).strip():
+        raise ValueError(
+            "validate_single_symbol_rule: missing single-symbol provenance; "
+            "fail closed, refusing an unprovenanced rule."
+        )
+    proven_symbol = str(provenance.get("symbol")).strip()
+    scoped: list[str] = []
+    for key in ("source_symbols", "island_symbols"):
+        raw = rule.get(key)
+        if raw is None:
+            continue
+        if isinstance(raw, str):
+            raw = [raw]
+        for entry in list(raw):
+            text = str(entry).strip()
+            if text and text not in scoped:
+                scoped.append(text)
+    if len(scoped) != 1:
+        raise ValueError(
+            f"validate_single_symbol_rule: expected exactly one "
+            f"single-symbol scope, got {scoped!r}; fail closed, refusing a "
+            "multi-symbol rule in specialist mode."
+        )
+    if scoped[0] != proven_symbol:
+        raise ValueError(
+            "validate_single_symbol_rule: provenance symbol "
+            f"{proven_symbol!r} disagrees with rule scope {scoped!r}; "
+            "fail closed, provenance must never be rewritten."
+        )
+    origin = rule.get("origin_symbol")
+    if origin is not None and str(origin).strip() != proven_symbol:
+        raise ValueError(
+            "validate_single_symbol_rule: origin_symbol "
+            f"{str(origin).strip()!r} disagrees with provenance symbol "
+            f"{proven_symbol!r}; fail closed, provenance must never be "
+            "rewritten."
+        )
+    if expected_symbol is not None and proven_symbol != str(
+        expected_symbol
+    ).strip():
+        raise ValueError(
+            f"validate_single_symbol_rule: cross-symbol migration from "
+            f"{proven_symbol!r} to {str(expected_symbol).strip()!r} is "
