+def complete_target_fixture() -> pd.DataFrame:
+    """Create a complete target DataFrame with legacy rule features."""
+    frame = target_fixture(pd.date_range("2024-02-01", periods=8, freq="15min"))
+    allowed_features = getattr(

+        frame[name] = float(position % 5)

+def complete_feature_frames() -> dict[str, pd.DataFrame]:
+    """Create a complete feature frame dictionary with all twelve on-chain features."""
+    frame = pd.DataFrame({
+        "available_at": [pd.Timestamp("2024-01-31", tz="UTC")],
+        **{name: [0.25] for name in ONCHAIN_FEATURE_NAMES},
+    })
+    return {"BTCUSDT": frame}

+def write_target(path: Path) -> Path:
+    """Write complete target fixture to CSV."""
+    complete_target_fixture().to_csv(path, index=False)
+    return path

+def write_complete_source_fixture(tmp_path: Path) -> dict[tuple[str, str], Path]:
+    """Write complete 24-file raw Santiment source fixtures for BTC and ETH."""
+    tmp_path.mkdir(parents=True, exist_ok=True)
+    paths: dict[tuple[str, str], Path] = {}
+    assets = (("Bitcoin", "BTCUSDT"), ("Ethereum", "ETHUSDT"))
+    for asset_name, symbol in assets:
+        for position, spec in enumerate(ONCHAIN_METRIC_SPECS, start=1):
+            cadence = "15min" if spec.source_cadence == "15min" else "1D"
+            periods = 16 if cadence == "15min" else 10
+            index = pd.date_range("2023-12-01", periods=periods, freq=cadence, tz="UTC")
+            values = np.arange(1, periods + 1, dtype=float) + position
+            path = tmp_path / f"{asset_name} metric-{position}.csv"
+            pd.DataFrame({"Date": index, spec.source_column: values}).to_csv(
+                path, index=False
+            )
+            paths[(symbol, spec.source_column)] = path
+    return paths

+def test_feature_is_not_visible_before_available_at() -> None:
+    """Test backward as-of join hides future data before available_at."""
+    target = target_fixture([
+        "2024-01-02 01:45",
+        "2024-01-02 02:00",
+        "2024-01-02 02:15",
+    ])
+    features = feature_fixture(available_at="2024-01-02 02:00", value=0.5)
+    enriched, _ = enrich_target_frame(target, {"BTCUSDT": features})
+    assert np.isnan(enriched.loc[0, "ff_oc_funding_rate"])
+    assert enriched.loc[1:, "ff_oc_funding_rate"].eq(0.5).all()

+def test_enrichment_preserves_source_and_adds_exactly_twelve_columns() -> None:
+    """Test enrichment preserves all original columns, rows, order and adds 12 features."""
+    source = complete_target_fixture()
+    enriched, audit = enrich_target_frame(source, complete_feature_frames())
+    pd.testing.assert_frame_equal(
+        enriched[source.columns], source, check_dtype=False
+    )
+    assert set(enriched.columns) - set(source.columns) == set(ONCHAIN_FEATURE_NAMES)
+    assert len(enriched) == len(source)
+    assert audit["duplicate_target_keys"] == 0

+def test_enrichment_rejects_duplicate_target_keys() -> None:
+    """Test enrichment fails closed when target DataFrame contains duplicate keys."""
+    source = complete_target_fixture()
+    duplicated = pd.concat([source, source.iloc[[0]]], ignore_index=True)
+    with pytest.raises(ValueError, match="duplicate target keys"):
+        enrich_target_frame(duplicated, complete_feature_frames())

+def test_enrichment_preserves_interleaved_symbol_order() -> None:
+    """Test enrichment preserves original row order across multiple interleaved symbols."""
+    btc_target = complete_target_fixture()
+    eth_target = complete_target_fixture()
+    eth_target["symbol"] = "ETHUSDT"
+
+    # Interleave rows: BTC row 0, ETH row 0, BTC row 1, ETH row 1...
+    interleaved_rows = []
+    for i in range(len(btc_target)):
+        interleaved_rows.append(btc_target.iloc[[i]])
+        interleaved_rows.append(eth_target.iloc[[i]])
+    interleaved = pd.concat(interleaved_rows, ignore_index=True)
+
+    features = {
+        "BTCUSDT": pd.DataFrame({
+            "available_at": [pd.Timestamp("2024-01-31", tz="UTC")],
+            **{name: [0.25] for name in ONCHAIN_FEATURE_NAMES},
+        }),
+        "ETHUSDT": pd.DataFrame({
+            "available_at": [pd.Timestamp("2024-01-31", tz="UTC")],
+            **{name: [-0.5] for name in ONCHAIN_FEATURE_NAMES},
+        }),
+    }
+
+    enriched, _ = enrich_target_frame(interleaved, features)
+    pd.testing.assert_frame_equal(
+        enriched[interleaved.columns], interleaved, check_dtype=False
+    )
+    assert (enriched.iloc[0::2]["symbol"] == "BTCUSDT").all()
+    assert (enriched.iloc[1::2]["symbol"] == "ETHUSDT").all()
+    assert enriched.iloc[0::2]["ff_oc_funding_rate"].eq(0.25).all()
+    assert enriched.iloc[1::2]["ff_oc_funding_rate"].eq(-0.5).all()

+def test_failed_build_never_replaces_original(tmp_path: Path, monkeypatch) -> None:
+    """Test atomic safety: failed staged verification leaves original file untouched."""
+    write_complete_source_fixture(tmp_path / "sources")
+    train = write_target(tmp_path / "train_new.csv")
+    before = train.read_bytes()
+
+    def fail_verification(*_args, **_kwargs):
+        raise ValueError("forced staged verification failure")
+
+    monkeypatch.setattr(builder, "verify_staged_dataset", fail_verification)
+    with pytest.raises(ValueError, match="forced staged verification failure"):
+        builder.main([
+            "--source-dir", str(tmp_path / "sources"),
+            "--train", str(train),
+            "--test", str(train),
+            "--output-train", str(train),
+            "--output-test", str(tmp_path / "test.csv"),
+            "--manifest", str(tmp_path / "manifest.json"),
+            "--replace",
+        ])
+    assert train.read_bytes() == before

+def test_verify_staged_dataset_detects_range_violation() -> None:
+    """Test staged verification rejects finite feature values outside [-1, 1]."""
+    source = complete_target_fixture()
+    enriched, _ = enrich_target_frame(source, complete_feature_frames())
+
+    # Range violation > 1.0
+    bad_enriched = enriched.copy()
+    bad_enriched.loc[0, "ff_oc_funding_rate"] = 1.5
+    with pytest.raises(ValueError, match="finite range violation"):
+        verify_staged_dataset(source, bad_enriched)
+
+    # Range violation < -1.0
+    bad_enriched2 = enriched.copy()
+    bad_enriched2.loc[0, "ff_oc_funding_rate"] = -1.05
+    with pytest.raises(ValueError, match="finite range violation"):
+        verify_staged_dataset(source, bad_enriched2)
